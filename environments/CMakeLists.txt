cmake_minimum_required(VERSION 3.18)
project(cybr_scene_shared_native LANGUAGES CXX)
include(CTest)
if(NOT MSVC)
  find_package(OpenMP REQUIRED)
endif()
function(scene_native_target name source)
  add_executable(${name} ${source})
  target_compile_features(${name} PRIVATE cxx_std_17)
  if(name MATCHES "test_")
    if(MSVC)
      target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
  endif()
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/shared")
  if(MSVC)
    target_compile_options(${name} PRIVATE /O2 /EHsc /openmp:llvm /arch:AVX2 /fp:precise)
    target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  else()
    target_compile_options(${name} PRIVATE -O3 -march=native -fno-math-errno)
    target_link_libraries(${name} PRIVATE OpenMP::OpenMP_CXX)
  endif()
endfunction()
scene_native_target(geo_r2 engines/geo/cybr-geo/native/spectral_scenes.cpp)
scene_native_target(hot_r2 engines/hot/cybr-geo/native/spectral_desert.cpp)

target_include_directories(obsidian_r2 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/engines/obsidian/include")
if(NOT MSVC)
  target_compile_options(geode_r2 PRIVATE -ffp-contract=off)
endif()
