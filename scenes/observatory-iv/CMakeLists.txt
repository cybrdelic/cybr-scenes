cmake_minimum_required(VERSION 3.16)
project(cybr_observatory_v4 LANGUAGES CXX)
find_package(OpenMP REQUIRED)
add_executable(observatory src/observatory.cpp)
target_compile_features(observatory PRIVATE cxx_std_17)
target_compile_options(observatory PRIVATE -O3 -mavx2 -mfma -Wall -Wextra -Wno-misleading-indentation)
target_link_libraries(observatory PRIVATE OpenMP::OpenMP_CXX)
enable_testing()
add_test(NAME transport_numeric COMMAND observatory --self-test)
